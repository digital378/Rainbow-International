  Kalwa   : { oneTime: 0,    annual: 60000 },

};
Key revenue metrics
Metric	Definition
Recognised Revenue	Σ(one-time + annual) for Adm Done rows in range
Projected Revenue	Σ expected revenue for OPEN DMs × confidence weight (High 70 %, Medium 40 %, Low 10 %)
CAC	Marketing Spend / Adm Done
ROI %	(Recognised Revenue − Spend) / Spend × 100
Payback Months	Months until cumulative revenue ≥ cumulative spend

Implementation notes

helpers/calcRevenue.ts

export function feeFor(branch:string){
  return (FEE_RULES[branch]?.oneTime ?? FEE_RULES.default.oneTime) +
         (FEE_RULES[branch]?.annual   ?? FEE_RULES.default.annual);
}
export function recognisedRevenue(leads:LeadRow[]){
  return leads.filter(r=>r.status==='Adm Done')
              .reduce((s,r)=>s+feeFor(r.branch),0);
}
export function projectedRevenue(openDm:LeadRow[]){
  const w = {High:0.7, Medium:0.4, Low:0.1} as const;
  return openDm.reduce((s,r)=>s+feeFor(r.branch)*w[r.salesConfidence],0);
}
