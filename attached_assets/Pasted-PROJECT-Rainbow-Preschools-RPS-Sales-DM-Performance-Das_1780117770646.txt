3.  B U G  F I X  — Median / P90 Lead-Time
---------------------------------------------------------------------------
Issue (v2.1): median shows 0 days due to rows missing both dates or date-order
errors.

Implementation:
```ts
// utils/metrics.ts
export function getLeadTimeStats(rows: LeadRow[]): {
  median: number; p90: number; sample: number; total: number
} {
  const deltas = rows
    .filter(r => r.enqDate && r.admDate && dayjs(r.admDate).isAfter(r.enqDate))
    .map(r => dayjs(r.admDate).diff(r.enqDate, 'day'));
  const median = d3.median(deltas) ?? 0;
  const p90    = d3.quantile(deltas, 0.9) ?? 0;
  return { median, p90, sample: deltas.length, total: rows.length };
}
